import os
import feedparser
from telegram import Update
from telegram.ext import Application, CommandHandler, MessageHandler, filters, ContextTypes
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

BOT_TOKEN = os.getenv("BOT_TOKEN")

if not BOT_TOKEN:
    print("ERROR: BOT_TOKEN not set!")
    exit(1)

saved_searches = {}

class IndeedJobBot:
    def search_jobs(self, job_title, location):
        jobs = []
        try:
            url = f"https://www.indeed.com/rss?q={job_title}&l={location}&limit=25"
            feed = feedparser.parse(url)
            
            for entry in feed.entries:
                job = {
                    'title': entry.title,
                    'link': entry.link,
                }
                jobs.append(job)
        except Exception as e:
            logger.error(f"Error: {e}")
            return []
        
        return jobs

job_bot = IndeedJobBot()
app = Application.builder().token(BOT_TOKEN).build()

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text(
        "👋 Welcome to Indeed Job Bot!\n\n"
        "/addsearch - Add job search\n"
        "/mysearches - View searches\n"
        "/removesearch - Remove search\n"
        "/help - Help"
    )

async def add_search(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text("What job title?\nExample: Python Developer")
    context.user_data['action'] = 'add_job_title'

async def my_searches(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.message.chat_id
    if user_id not in saved_searches or not saved_searches[user_id]:
        await update.message.reply_text("❌ No searches yet.\nType /addsearch")
        return
    
    message = "📋 Your searches:\n\n"
    for i, search in enumerate(saved_searches[user_id], 1):
        message += f"{i}. {search['job']} - {search['location']}\n"
    
    await update.message.reply_text(message)

async def remove_search(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.message.chat_id
    if user_id not in saved_searches or not saved_searches[user_id]:
        await update.message.reply_text("❌ No searches to remove.")
        return
    
    message = "Which search to remove?\n\n"
    for i, search in enumerate(saved_searches[user_id], 1):
        message += f"{i}. {search['job']} - {search['location']}\n"
    
    message += "\nReply with number"
    await update.message.reply_text(message)
    context.user_data['action'] = 'remove_search'

async def help_command(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text(
        "📖 How to use:\n\n"
        "/addsearch - Add search\n"
        "/mysearches - View\n"
        "/removesearch - Remove"
    )

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.message.chat_id
    user_input = update.message.text.strip()
    action = context.user_data.get('action')
    
    if action == 'add_job_title':
        context.user_data['job_title'] = user_input
        await update.message.reply_text("Where? (Manchester, London, Remote)")
        context.user_data['action'] = 'add_location'
        return
    
    if action == 'add_location':
        location = user_input
        job_title = context.user_data.get('job_title')
        
        if user_id not in saved_searches:
            saved_searches[user_id] = []
        
        saved_searches[user_id].append({
            'job': job_title,
            'location': location
        })
        
        await update.message.reply_text(
            f"✅ Added!\nJob: {job_title}\nLocation: {location}"
        )
        context.user_data['action'] = None
        return
    
    if action == 'remove_search':
        try:
            index = int(user_input) - 1
            if user_id in saved_searches and 0 <= index < len(saved_searches[user_id]):
                removed = saved_searches[user_id].pop(index)
                await update.message.reply_text(f"✅ Removed: {removed['job']}")
            else:
                await update.message.reply_text("❌ Invalid number")
        except:
            await update.message.reply_text("❌ Enter a number")
        context.user_data['action'] = None

app.add_handler(CommandHandler("start", start))
app.add_handler(CommandHandler("addsearch", add_search))
app.add_handler(CommandHandler("mysearches", my_searches))
app.add_handler(CommandHandler("removesearch", remove_search))
app.add_handler(CommandHandler("help", help_command))
app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))

if __name__ == '__main__':
    print("🤖 Bot starting...")
    app.run_polling()
